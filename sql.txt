-- Create the database (execute this command in pgAdmin's query tool)
CREATE DATABASE library_dbms;

-- Connect to the newly created database
\c library_dbms

-- Create tables
CREATE TABLE users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(50) NOT NULL
);

CREATE TABLE students (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    branch VARCHAR(50) NOT NULL,
    year INTEGER NOT NULL,
    ug_pg VARCHAR(10) NOT NULL,
    photo BYTEA
);

CREATE TABLE faculty (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    department VARCHAR(100) NOT NULL,
    photo BYTEA,
    steak_of_days INTEGER
);

CREATE TABLE logs (
    id SERIAL PRIMARY KEY,
    person_id INTEGER NOT NULL,
    person_type VARCHAR(50) NOT NULL,
    check_in_time TIMESTAMP,
    check_out_time TIMESTAMP,
    FOREIGN KEY (person_id) REFERENCES students (id),
    FOREIGN KEY (person_id) REFERENCES faculty (id)
);
